#![windows_subsystem = "windows"]
// Throwaway spike for SPEC-0200-0001 §3.16 S-1..S-4. Logs to spike.log next to the exe.
use std::fs::OpenOptions;
use std::io::Write;
use std::os::windows::fs::MetadataExt;
use std::os::windows::process::CommandExt;
use std::path::PathBuf;
use std::process::{Command, Stdio};

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let mode = args.get(1).cloned().unwrap_or_default();
    let tmp = PathBuf::from(args.get(2).cloned().unwrap_or_default());
    let mut log = OpenOptions::new().create(true).append(true)
        .open(std::env::current_exe().unwrap().with_file_name("spike.log")).unwrap();
    let mut w = |s: String| { writeln!(log, "{s}").unwrap(); };
    match mode.as_str() {
        "s1-meta" => {
            let p = PathBuf::from(std::env::var("LOCALAPPDATA").unwrap()).join(r"Microsoft\WindowsApps\wt.exe");
            match std::fs::symlink_metadata(&p) {
                Ok(m) => w(format!("S1 symlink_metadata ok: attrs=0x{:x} is_file={} is_symlink={} len={}", m.file_attributes(), m.is_file(), m.file_type().is_symlink(), m.len())),
                Err(e) => w(format!("S1 symlink_metadata err: {e}")),
            }
            match std::fs::metadata(&p) {
                Ok(m) => w(format!("S1 metadata(follow) ok: attrs=0x{:x} is_file={}", m.file_attributes(), m.is_file())),
                Err(e) => w(format!("S1 metadata(follow) err: {e}")),
            }
            w(format!("S1 Path::exists={} is_file={}", p.exists(), p.is_file()));
        }
        "s1-launch" => {
            let p = PathBuf::from(std::env::var("LOCALAPPDATA").unwrap()).join(r"Microsoft\WindowsApps\wt.exe");
            let r = Command::new(&p).arg("-d").arg(&tmp).current_dir(&tmp)
                .stdin(Stdio::null()).stdout(Stdio::null()).stderr(Stdio::null()).spawn();
            w(format!("S1 launch wt -d {:?}: {:?}", tmp, r.map(|c| c.id())));
        }
        "s2" => {
            let target = tmp.join("a b,c.txt");
            let explorer = PathBuf::from(std::env::var("SystemRoot").unwrap()).join("explorer.exe");
            let arg = format!("/select,{}", target.display());
            let r = Command::new(&explorer).arg(&arg).spawn();
            w(format!("S2 explorer {:?}: {:?}", arg, r.map(|c| c.id())));
        }
        "s2b" | "s2c" | "s2d" => {
            let explorer = PathBuf::from(std::env::var("SystemRoot").unwrap()).join("explorer.exe");
            let mut c = Command::new(&explorer);
            let desc;
            if mode == "s2b" { let t = tmp.join("a b,c.txt"); desc = format!("raw /select,\"{}\"", t.display()); c.raw_arg(format!("/select,\"{}\"", t.display())); }
            else if mode == "s2c" { let t = tmp.join("ab.txt"); desc = format!("arg /select,{} (no space dir? see tmp)", t.display()); c.arg(format!("/select,{}", t.display())); }
            else { let t = tmp.join("a b,c.txt"); desc = format!("args [/select, path]"); c.arg("/select,").arg(&t); }
            let r = c.spawn();
            w(format!("{mode} explorer {desc}: {:?}", r.map(|c| c.id())));
        }
        "s3-null" | "s3-inherit" => {
            let cmd = PathBuf::from(std::env::var("SystemRoot").unwrap()).join(r"System32\cmd.exe");
            let mut c = Command::new(&cmd);
            c.current_dir(&tmp).creation_flags(0x10); // CREATE_NEW_CONSOLE
            if mode == "s3-null" { c.stdin(Stdio::null()).stdout(Stdio::null()).stderr(Stdio::null()); }
            else { c.stdout(Stdio::null()).stderr(Stdio::null()); }
            let r = c.spawn();
            w(format!("S3 {mode} cmd.exe new console in {:?}: {:?}", tmp, r.map(|c| c.id())));
        }
        "s4" => {
            // long working folder (> 260 chars) and a trailing-backslash UNC-like arg (not launched)
            let mut long = tmp.clone();
            while long.as_os_str().len() < 300 { long.push("very_long_folder_name_for_spike_0123456789"); }
            std::fs::create_dir_all(&long).ok();
            let cmd = PathBuf::from(std::env::var("SystemRoot").unwrap()).join(r"System32\cmd.exe");
            let r = Command::new(&cmd).args(["/c", "cd"]).current_dir(&long)
                .stdout(Stdio::null()).stderr(Stdio::null()).stdin(Stdio::null()).creation_flags(0x08000000).status();
            w(format!("S4 long cwd len={} status: {:?}", long.as_os_str().len(), r));
        }
        _ => w(format!("unknown mode {mode}")),
    }
}
